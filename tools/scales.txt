# FXKit scales that JavaFX CSS cannot hold as tokens (see docs/css-notes.md, section 8).
# tools/GenerateUtilities.java turns them into utility classes in utilities.css.
#
# Format:  <family> <name> <value>
#   space   name  px      -> .p-<name>, .px-<name>, .py-<name>, .pt/.pr/.pb/.pl-<name>, .gap-<name>
#   radius  name  px      -> .rounded-<name>      (name "base" gives plain .rounded)
#   font    name  px      -> .text-<name>
#   weight  name  value   -> .font-<name>         (CSS keyword or 100-900)
#
# To change a value, edit it here and run:  java tools/GenerateUtilities.java
